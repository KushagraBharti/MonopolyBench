% Fable proposal: link and unit macros for the shared package.
% \exhibitref{F4}{astra/exhibits/f04_opportunity_and_deployment.pdf}
%   prints the link text and a footnote with the repository-relative path.
%   No PDF action is emitted (hyperref's run: prefix produces /S/Launch,
%   which viewers block and which resolves relative to the viewer's cwd).
\newcommand{\exhibitref}[2]{#1\footnote{Repository path: \texttt{\detokenize{#2}}}}
% Internal reference for exhibits that live in the same PDF.
\newcommand{\figref}[1]{Figure~\ref{#1}}
\newcommand{\tabref}[1]{Table~\ref{#1}}
% Two currencies that must never share a macro.
\newcommand{\gd}[1]{\$#1}              % game currency, e.g. \gd{103,450}
\newcommand{\usd}[1]{USD~#1}           % provider spend from OpenRouter, e.g. \usd{141.71}
% Verbatim canonical tokens (event codes, property ids): no hyphenation, no subscripts.
\newcommand{\tok}[1]{\mbox{\texttt{\detokenize{#1}}}}
